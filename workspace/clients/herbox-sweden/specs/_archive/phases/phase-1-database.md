### `app/models/pending_orders.py`

```python
"""
Pending order models for the Order Approval Dashboard.

PRD: specs/prd-order-dashboard.md
"""

import uuid
from datetime import datetime, date
from decimal import Decimal

from sqlalchemy import (
    Column, String, DateTime, Date, JSON, Integer, Text,
    Numeric, ForeignKey, Index
)
from sqlalchemy.dialects.postgresql import UUID
from sqlalchemy.orm import relationship

from ..db import Base


class PendingOrder(Base):
    """Order generated by A1 workflow, awaiting approval."""

    __tablename__ = "pending_orders"

    id = Column(UUID(as_uuid=True), primary_key=True, default=uuid.uuid4)

    # Source tracking
    contract_number = Column(String(50), nullable=False)
    customer_number = Column(String(50), nullable=False, index=True)
    customer_name = Column(String(255))
    source = Column(String(20), nullable=False, default="recurring")  # 'recurring' | 'new'

    # Full Fortnox Order payload (ready to POST)
    order_payload = Column(JSON, nullable=False)

    # Denormalized display fields (for fast table rendering)
    delivery_date = Column(Date, nullable=False)
    order_date = Column(Date)
    total_amount = Column(Numeric(12, 2))
    currency = Column(String(10), default="SEK")
    item_count = Column(Integer)
    item_summary = Column(Text)  # e.g. "3x Mensskydd Tena, 2x Tvattlapp"

    # Enrichment fields
    administration_fee = Column(Numeric(12, 2), default=0)
    freight = Column(Numeric(12, 2), default=0)
    remarks = Column(Text)
    period_start = Column(Date)

    # Workflow state
    status = Column(String(20), nullable=False, default="pending", index=True)
    # Values: 'pending', 'approved', 'denied', 'created', 'failed'

    # Duplicate prevention
    your_order_number = Column(String(100), unique=True, nullable=False, index=True)

    # Post-creation tracking
    fortnox_order_number = Column(String(50))
    error_message = Column(Text)

    # Timestamps
    generated_at = Column(DateTime, default=datetime.utcnow)
    reviewed_at = Column(DateTime)
    created_at = Column(DateTime, default=datetime.utcnow)
    updated_at = Column(DateTime, default=datetime.utcnow, onupdate=datetime.utcnow)

    # Relationships
    approval_logs = relationship("ApprovalLog", back_populates="order", order_by="ApprovalLog.created_at.desc()")

    __table_args__ = (
        Index("idx_pending_orders_status_source", "status", "source"),
        Index("idx_pending_orders_delivery_date", "delivery_date"),
    )


class ApprovalLog(Base):
    """Audit trail for all actions on pending orders."""

    __tablename__ = "approval_log"

    id = Column(UUID(as_uuid=True), primary_key=True, default=uuid.uuid4)
    order_id = Column(UUID(as_uuid=True), ForeignKey("pending_orders.id"), nullable=False, index=True)
    action = Column(String(20), nullable=False)  # 'approved', 'denied', 'edited', 'created', 'failed'
    performed_by = Column(String(100))
    details = Column(JSON)  # What changed (for edits)
    created_at = Column(DateTime, default=datetime.utcnow)

    # Relationships
    order = relationship("PendingOrder", back_populates="approval_logs")
```

## Files to Modify

### `app/db.py`

Add import of new models so they're registered with `Base.metadata` before `init_db()` is called:

```python
# Add at the bottom of the file, before init_db():
from .models.pending_orders import PendingOrder, ApprovalLog  # noqa: F401, E402
```

This ensures `Base.metadata.create_all()` in `init_db()` picks up the new tables.

### `app/main.py`

No changes needed — `init_db()` already calls `Base.metadata.create_all()` in the `lifespan()` function (line 43), which will automatically create the new tables on next startup.

---

## Migration Strategy

Since this is a new table (not modifying existing ones), we use the existing `init_db()` auto-creation pattern:

1. **Local:** Tables created automatically when app starts (SQLite or Postgres)
2. **Production (Railway):** Tables created on next deployment when `init_db()` runs

No Alembic migration needed — the app already uses `Base.metadata.create_all()` which is additive (creates tables that don't exist, leaves existing tables untouched).
